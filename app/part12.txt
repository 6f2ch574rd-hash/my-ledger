)-1,1);
    return[fmtYM(d)+'-01',fmtYMD(new Date(y,now.getMonth(),0))];}
  if(statRange==='year')return[y+'-01-01',today()];
  if(statRange==='custom')return[$('rStart').value||'2000-01-01',$('rEnd').value||today()];
  return['2000-01-01','2999-12-31'];
}
function switchPie(t){haptic();selPieCat=null;pieType=t;
  $('pieExpBtn').style.background=t==='expense'?'var(--wine)':'#f0f0f2';
  $('pieExpBtn').style.color=t==='expense'?'#fff':'#333';
  $('pieIncBtn').style.background=t==='income'?'var(--green)':'#f0f0f2';
  $('pieIncBtn').style.color=t==='income'?'#fff':'#333';
  renderStats();
}
function pieClick(cat){
  haptic();
  selPieCat=(!cat||cat===selPieCat)?null:cat;
  renderStats();
}
function renderStats(){
  const [s,e]=statPeriod();
  const rows=D().tx.filter(r=>r.date>=s&&r.date<=e);
  let exp=0,inc=0;const eMap={},iMap={};
  rows.forEach(r=>{
    if(r.type==='expense'){exp+=r.amount;eMap[r.category]=(eMap[r.category]||0)+r.amount;}
    else{inc+=r.amount;iMap[r.category]=(iMap[r.category]||0)+r.amount;}
  });
  $('stExp').textContent=fmt(exp);$('stInc').textContent=fmt(inc);$('stBal').textContent=fmt(inc-exp);
  const catMap=pieType==='expense'?eMap:iMap;
  const entries=Object.entries(catMap).sort((a,b)=>b[1]-a[1]);
  const total=entries.reduce((s,x)=>s+x[1],0);
  const colors=['#9E203E','#E8A0A8','#E67E5A','#E8C56A','#4A8B6F','#6A8FBF','#9B7FBF','#D9B38C','#C0506A','#7FA88C','#B5915A','#8E7CC3'];
  if(!entries.length){$('donutBox').innerHTML='<div class="empty">這個區間沒有資料</div>';$('rankBox').innerHTML='';}
  else{
    const R=88,C=2*Math.PI*R,cx=110,cy=110;let acc=0;
    const arcs=entries.map((en,i)=>{
      const frac=en[1]/total,len=frac*C,gap=entries.length>1?2:0;
      const dim=selPieCat&&selPieCat!==en[0]?' opacity="0.28"':'';
      const seg=`<circle cx="${cx}" cy="${cy}" r="${R}" fill="none" stroke="${colors[i%colors.length]}"
        stroke-width="34" stroke-dasharray="${Math.max(len-gap,0)} ${C-len+gap}" stroke-dashoffset="${-acc*C}"
        transform="rotate(-90 ${cx} ${cy})"${dim} onclick="pieClick('${esc(en[0])}')">
        <title>${esc(en[0])} ${fmt(en[1])}（${(frac*100).toFixed(1)}%）</title></circle>`;
      acc+=frac;return seg;
    }).join('');
    let center;
    if(selPieCat){
      const en=entries.find(x=>x[0]===selPieCat);

      center=`<text x="${cx}" y="${cy-14}" text-anchor="middle" font-size="12" fill="#888">${esc(catName(selPieCat))}</text>
        <text x="${cx}" y="${cy+6}" text-anchor="middle" font-size="15" font-weight="800" fill="${pieType==='expense'?'#9E203E':'#27AE60'}">${fmt(en?en[1]:0).replace('NT$','')}</text>
        <text x="${cx}" y="${cy+22}" text-anchor="middle" font-size="10" fill="#888">${en?((en[1]/total*100).toFixed(1)):'0'}%・${cnt}筆・點擊返回</text>`;
    }else{

        <text x="${cx}" y="${cy+16}" text-anchor="middle" font-size="17" font-weight="800" fill="${pieType==='expense'?'#9E203E':'#27AE60'}">${total>=10000?(total/10000).toFixed(1)+'萬':Math.round(total).toLocaleString()}</text>
        <text x="${cx}" y="${cy+34}" text-anchor="middle" font-size="9" fill="#aaa">點圓環看分類詳情</text>`;
    }

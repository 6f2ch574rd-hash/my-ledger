function renderHome(){
  renderBookSelect();
  const d=D(),m=fmtYM(new Date());
  const rows=d.tx.filter(r=>r.date.startsWith(m));
  let exp=0,inc=0;rows.forEach(r=>r.type==='expense'?exp+=r.amount:inc+=r.amount);
  $('heroMonth').textContent=new Date().getFullYear()+' 年 '+(new Date().getMonth()+1)+' 月';
  $('heroExp').textContent=fmt(exp);$('heroInc').textContent=fmt(inc);$('heroBal').textContent=fmt(inc-exp);
  if(d.budget>0){
    const pct=exp/d.budget;
    const bar=$('budgetBar');bar.style.width=Math.min(pct*100,100)+'%';
    bar.className=pct>1?'over':pct>0.8?'warn':'';
    $('budgetText').textContent=fmt(exp)+' / '+fmt(d.budget);
    $('budgetHint').textContent=pct>1?`已超出預算 ${fmt(exp-d.budget)}`:`還可使用 ${fmt(Math.max(d.budget-exp,0))}（${(pct*100).toFixed(0)}%）`;
    $('budgetCard').style.display='';
  }else $('budgetCard').style.display='none';
  $('acctBox').innerHTML=d.accounts.map(a=>{
    let bal=a.opening||0;
    d.tx.forEach(r=>{if(r.account!==a.id)return;bal+=r.type==='income'?r.amount:-r.amount;});
    return `<div class="acct-row"><span>${a.icon} ${esc(a.name)}</span><b>${fmt(bal)}</b></div>`;
  }).join('')||'<div class="empty">尚無帳戶</div>';
  const recent=[...d.tx].sort((a,b)=>b.date.localeCompare(a.date)||b.createdAt-a.createdAt).slice(0,7);
  $('recentBox').innerHTML=recent.length?recent.map(txRow).join(''):'<div class="empty"><span class="big-e">🧾</span>還沒有記錄，點下方＋開始記第一筆</div>';
}
function txRow(r){
  const acct=D().accounts.find(a=>a.id===r.account);
  return `<div class="tx" onclick="editTx('${r.id}')">
    <div class="emoji ${r.type==='income'?'inc':''}">${catEmoji(r.category)}</div>
    <div class="info"><div class="n">${esc(r.name)}</div>
    <div class="m">${r.date} · ${esc(catName(r.category))}${acct?' · '+acct.icon+esc(acct.name):''}</div></div>
    <div class="amt ${r.type==='expense'?'exp':'inc'}">${r.type==='expense'?'-':'+'}${fmt(r.amount).replace('NT$','')}</div>
  </div>`;
}

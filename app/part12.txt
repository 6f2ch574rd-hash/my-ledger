function doSave(){
  haptic();
  const d=D();
  const amount=Number($('fAmount').value.replace(/[^\d.]/g,''));
  if(!amount||amount<=0){toast('請輸入正確金額');return;}
  const cats=curType==='expense'?d.catsExp:d.catsInc;
  const cat=selCat||cats[0]||'🧾 其他';
  const acct=$('fAccount').value||d.accounts[0]?.id||'';
  const rec={
    id:editingId||uid(),date:$('fDate').value||today(),
    name:$('fName').value.trim()||catName(cat),
    category:cat,amount:amount,type:curType,account:acct,
    note:$('fNote').value.trim(),
    createdAt:editingId?(d.tx.find(x=>x.id===editingId)?.createdAt||nowTs()):nowTs()
  };
  if(editingId){const i=d.tx.findIndex(x=>x.id===editingId);if(i>=0)d.tx[i]=rec;toast('已更新這筆記錄');}
  else{d.tx.push(rec);toast(curType==='income'?'💰 收入已入帳':'✅ 支出已記錄');}
  saveDB();renderAddForm();
}
function editTx(id){
  const r=D().tx.find(x=>x.id===id);if(!r)return;
  editingId=id;setType(r.type);
  selCat=r.category;renderCatGrid();
  $('fDate').value=r.date;$('fName').value=r.name===catName(r.category)?'':r.name;
  $('fAmount').value=r.amount;$('fNote').value=r.note||'';
  renderAccounts(r.account);
  $('editDelBtn').style.display='block';
  goPage('pageAdd',document.querySelector('.nav button[data-p="pageAdd"],.nav label[data-p="pageAdd"]'));
}
function deleteEditing(){
  haptic();
  if(!editingId)return;
  if(!confirm('確定刪除這筆記錄？'))return;
  D().tx=D().tx.filter(x=>x.id!==editingId);
  saveDB();editingId=null;renderAddForm();
  goPage('pageHome',document.querySelector('.nav button[data-p="pageHome"],.nav label[data-p="pageHome"]'));
  toast('已刪除');
}

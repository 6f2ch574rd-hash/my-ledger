function goPage(pid,btn){
  haptic();
  document.querySelectorAll('.nav button,.nav label[for]').forEach(b=>b.classList.remove('active'));
  if(typeof btn==='number')btn=document.querySelector(`.nav button[data-p="${pid}"],.nav label[data-p="${pid}"]`);
  if(btn&&btn.dataset)btn.classList.add('active');
  document.querySelectorAll('.page').forEach(p=>p.classList.remove('active'));
  $(pid).classList.add('active');
  window.scrollTo(0,0);
  if(pid==='pageHome')renderHome();
  if(pid==='pageList')renderList();
  if(pid==='pageStats')renderStats();
  if(pid==='pageSettings')renderSettings();
  if(pid==='pageAdd'&&!editingId)renderAddForm();
}

let curType='expense',selCat=null,editingId=null;
function setType(t){
  haptic();
  curType=t;const seg=$('typeSeg');
  seg.classList.toggle('inc',t==='income');
  seg.children[0].className=t==='expense'?'on-exp':'';
  seg.children[1].className=t==='income'?'on-inc':'';
  $('saveBtn').className='btn '+(t==='income'?'btn-green':'btn-wine');
  $('fAmount').className='amt-input'+(t==='income'?' inc':'');
  selCat=null;renderCatGrid();
}
function renderCatGrid(){
  const cats=curType==='expense'?D().catsExp:D().catsInc;
  $('catGrid').innerHTML=cats.map(c=>{
    const sel=selCat===c?(curType==='income'?'sel-inc':'sel'):'';
    return `<div class="cat-cell ${sel}" onclick="pickCat('${esc(c)}')"><span class="e">${catEmoji(c)}</span>${esc(catName(c))}</div>`;
  }).join('');
}
function pickCat(c){haptic();selCat=c;renderCatGrid();}
function renderAccounts(sel){
  $('fAccount').innerHTML=D().accounts.map(a=>`<option value="${a.id}" ${sel===a.id?'selected':''}>${a.icon} ${esc(a.name)}</option>`).join('');
}
function renderAddForm(){
  editingId=null;$('editDelBtn').style.display='none';
  $('fDate').value=today();$('fName').value='';$('fAmount').value='';$('fNote').value='';
  renderAccounts(D().accounts[0]?.id);setType('expense');
}
function resetForm(){renderAddForm();toast('表單已清空');}

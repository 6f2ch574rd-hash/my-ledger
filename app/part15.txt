          + (rowsSel.length>30?`<div class="mini" style="padding-top:6px;">僅顯示前 30 筆</div>`:'')}
        </div>`);
    }
  }
  const months=[];const now=new Date();
  for(let k=5;k>=0;k--)months.push(fmtYM(new Date(now.getFullYear(),now.getMonth()-k,1)));
  const mv=months.map(m=>D().tx.filter(r=>r.type==='expense'&&r.date.startsWith(m)).reduce((s,r)=>s+r.amount,0));
  const max=Math.max(...mv,1),W=320,H=105,top=38,bw=W/6;
  $('trendBox').innerHTML=`<svg viewBox="0 0 ${W} ${top+H+26}" style="width:100%;">`+
    mv.map((v,i)=>{const h=Math.max(v/max*H,v?5:0);
      const sel=selTrend===i,dim=selTrend!==null&&selTrend!==i;
      return `<rect x="${i*bw+bw*0.22}" y="${top+H-h}" width="${bw*0.56}" height="${h}" rx="4"
        fill="${i===5?'#9E203E':'#E8A0A8'}"${dim?' opacity="0.35"':''} style="cursor:pointer" onclick="trendClick(${i})">
        <title>${monthLabel(months[i])}：${fmt(v)}</title></rect>
        ${sel?`<text x="${i*bw+bw/2}" y="${top+H-h-7}" text-anchor="middle" font-size="11" font-weight="700" fill="${i===5?'#9E203E':'#b05a70'}">${v>=10000?(v/10000).toFixed(1)+'萬':Math.round(v).toLocaleString()}</text>`:''}
        <text x="${i*bw+bw/2}" y="${top+H+20}" text-anchor="middle" font-size="10" fill="#888">${Number(months[i].slice(5))}月</text>`;
    }).join('')+`</svg>`;
}

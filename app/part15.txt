.max(v/max*H,v?5:0);
      const sel=selTrend===i,dim=selTrend!==null&&selTrend!==i;
      return `<rect x="${i*bw+bw*0.22}" y="${top+H-h}" width="${bw*0.56}" height="${h}" rx="4"
        fill="${i===5?'#9E203E':'#E8A0A8'}"${dim?' opacity="0.35"':''} style="cursor:pointer" onclick="trendClick(${i})">
        <title>${monthLabel(months[i])}：${fmt(v)}</title></rect>
        ${sel?`<text x="${i*bw+bw/2}" y="${top+H-h-7}" text-anchor="middle" font-size="11" font-weight="700" fill="${i===5?'#9E203E':'#b05a70'}">${v>=10000?(v/10000).toFixed(1)+'萬':Math.round(v).toLocaleString()}</text>`:''}
        <text x="${i*bw+bw/2}" y="${top+H+20}" text-anchor="middle" font-size="10" fill="#888">${Number(months[i].slice(5))}月</text>`;
    }).join('')+`</svg>`;
}
function trendClick(i){haptic();selTrend=(selTrend===i)?null:i;renderStats();}

function openBookMgr(){renderBookMgr();$('bookSheet').classList.add('show');}
function closeSheet(id){$(id).classList.remove('show');}
function bookRowsHtml(){
  return books.map(b=>`
    <div class="acct-row"><span>📓 ${esc(b.name)}${b.id===activeId?' <b class="mini" style="color:var(--wine)">使用中</b>':''}</span>
      <span><button class="btn btn-sm btn-gray" onclick="useBook('${b.id}')">切換</button>
      <button class="del-x" onclick="removeBook('${b.id}')">刪</button></span></div>`).join('');
}
function renderBookMgr(){
  $('bookMgrList').innerHTML=bookRowsHtml();
  $('bookListBox').innerHTML=bookRowsHtml()+
    `<button class="btn btn-sm btn-wine" style="margin-top:10px;" onclick="createLocalBook()">＋ 新增一本帳</button>`;
}
function useBook(id){haptic();activeId=id;saveBooks();closeSheet('bookSheet');refreshAll();toast('已切換帳本');}
function createLocalBook(){
  haptic();
  const name=(prompt('新帳本名稱（例：旅行基金、小家庭）')||'').trim();if(!name)return;
  const b={id:uid(),name,createdAt:nowTs()};
  books.push(b);DB[b.id]=freshBookData();activeId=b.id;saveBooks();saveDB();
  closeSheet('bookSheet');refreshAll();toast('已建立帳本：'+name);
}
function removeBook(id){
  haptic();
  if(books.length===1){toast('至少保留一本帳');return;}
  if(!confirm('刪除這本帳會一併刪除其中所有記錄，確定？'))return;
  books=books.filter(b=>b.id!==id);delete DB[id];
  if(activeId===id){activeId=books[0].id;}
  saveBooks();saveDB();renderBookMgr();refreshAll();
}

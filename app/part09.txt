async function doBioUnlock(silent){
  if(!settings.bioCred){if(!silent)toast('尚未設定 Face ID');return;}
  try{
    const ch=new Uint8Array(32);crypto.getRandomValues(ch);
    await navigator.credentials.get({publicKey:{
      challenge:ch,
      rpId:location.hostname,
      allowCredentials:[{type:'public-key',id:b64urlToBytes(settings.bioCred)}],
      userVerification:'required',timeout:60000
    }});
    $('lockScreen').classList.remove('show');
  }catch(e){if(!silent)toast('Face ID 驗證失敗');haptic();}
}

let listCursor=fmtYM(new Date()),listFilter='all';
function shiftMonth(d){
  haptic();
  const [y,m]=listCursor.split('-').map(Number);
  listCursor=fmtYM(new Date(y,m-1+d,1));renderList();
}
function setFilter(f){haptic();listFilter=f;
  document.querySelectorAll('#filterChips .chip').forEach(c=>c.classList.toggle('on',c.dataset.f===f));
  renderList();
}
function renderList(){
  $('listMonth').textContent=monthLabel(listCursor);
  const kw=($('searchBox')?.value||'').trim().toLowerCase();
  let rows=D().tx.filter(r=>r.date.startsWith(listCursor));
  if(listFilter!=='all')rows=rows.filter(r=>r.type===listFilter);
  if(kw)rows=rows.filter(r=>(r.name+r.category+(r.note||'')).toLowerCase().includes(kw));
  rows.sort((a,b)=>b.date.localeCompare(a.date)||b.createdAt-a.createdAt);
  const box=$('listBox');
  if(!rows.length){box.innerHTML='<div class="empty"><span class="big-e">📭</span>這個月沒有記錄</div>';return;}
  const groups={};
  rows.forEach(r=>{(groups[r.date]=groups[r.date]||[]).push(r);});
  let html='';
  Object.keys(groups).sort().reverse().forEach(date=>{
    const list=groups[date];
    const de=list.filter(r=>r.type==='expense').reduce((s,r)=>s+r.amount,0);
    const di=list.filter(r=>r.type==='income').reduce((s,r)=>s+r.amount,0);
    const wd=['日','一','二','三','四','五','六'][new Date(date+'T12:00:00').getDay()];
    html+=`<div class="day-head"><span>${date.slice(5).replace('-','/')} 週${wd}</span>
      <span>${de?'支 '+de.toLocaleString():''}${de&&di?' · ':''}${di?'收 '+di.toLocaleString():''}</span></div>`;
    html+=list.map(txRow).join('');
  });
  box.innerHTML=html;
}

function statPeriod(){

  if(statRange==='month')return[fmtYM(now)+'-01',today()];
  if(statRange==='lastmonth'){const d=new Date(y,now.getMonth()-1,1);
    return[fmtYM(d)+'-01',fmtYMD(new Date(y,now.getMonth(),0))];}
  if(statRange==='year')return[y+'-01-01',today()];
  if(statRange==='custom')return[$('rStart').value||'2000-01-01',$('rEnd').value||today()];

}

function bioSupported(){
  return !!(window.isSecureContext&&navigator.credentials&&navigator.credentials.create
    &&navigator.credentials.get&&window.PublicKeyCredential!==undefined);
}
function b64urlToBytes(s){
  const b=atob(s.replace(/-/g,'+').replace(/_/g,'/'));
  const u=new Uint8Array(b.length);
  for(let i=0;i<b.length;i++)u[i]=b.charCodeAt(i);
  return u;
}
function bytesToB64url(u){
  let s='';const chunk=0x8000;
  for(let i=0;i<u.length;i+=chunk)s+=String.fromCharCode.apply(null,u.subarray(i,i+chunk));
  return btoa(s).replace(/\+/g,'-').replace(/\//g,'_').replace(/=+$/,'');
}
async function toggleBio(on){
  haptic();
  if(on){
    if(!bioSupported()){toast('此瀏覽器不支援 Face ID／Touch ID');$('bioSwitch').checked=false;return;}
    if(!settings.pass){toast('請先設定 4 位數密碼鎖');$('bioSwitch').checked=false;return;}
    if(settings.bioCred){settings.bio=true;LS.set('ll_settings',settings);renderSettings();toast('已啟用 Face ID 解鎖');return;}
    try{
      const ch=new Uint8Array(32);crypto.getRandomValues(ch);
      const uid=new Uint8Array(16);crypto.getRandomValues(uid);
      const cred=await navigator.credentials.create({publicKey:{
        challenge:ch,
        rp:{id:location.hostname,name:'我的記帳本'},
        user:{id:uid,name:settings.pass,displayName:'我的記帳本'},
        pubKeyCredParams:[{type:'public-key',alg:-7},{type:'public-key',alg:-257}],
        authenticatorSelection:{authenticatorAttachment:'platform',userVerification:'required',residentKey:'preferred'},
        timeout:60000,attestation:'none'
      }});
      settings.bioCred=cred.id;settings.bio=true;LS.set('ll_settings',settings);
      renderSettings();toast('✅ 已啟用 Face ID 解鎖');
    }catch(e){toast('啟用失敗（已取消或裝置不支援）');$('bioSwitch').checked=false;}
  }else{
    settings.bio=false;settings.bioCred=null;LS.set('ll_settings',settings);
    renderSettings();toast('已關閉 Face ID 解鎖');
  }
}

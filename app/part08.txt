/* ================= 觸覺反饋（震動＋點擊音） =================
   全部由設定頁「震動提示」單一開關控制：關閉即整組停用；
   未來 iOS 若封鎖此機制，只需移除下方 haptic-trigger 元素＋本函數即可。 */
let _tickCtx=null;
function playTick(){
  try{
    if(!_tickCtx)_tickCtx=new (window.AudioContext||window.webkitAudioContext)();
    if(_tickCtx.state==='suspended')_tickCtx.resume();
    const o=_tickCtx.createOscillator(),g=_tickCtx.createGain();
    o.type='sine';o.frequency.value=1900;
    g.gain.value=0.03;
    o.connect(g);g.connect(_tickCtx.destination);
    o.start();o.stop(_tickCtx.currentTime+0.014);
  }catch(e){}
}
function haptic(){
  if(!settings.haptic)return;
  playTick();                       // 聽覺回饋：14ms 輕點音
  const t=document.getElementById('haptic-trigger');
  if(t){try{t.click();}catch(e){}}  // iOS 17.4+：隱藏 switch 觸發 Taptic
  try{if(navigator.vibrate)navigator.vibrate(12);}catch(e){}
  // 雙連觸發：40ms 後再補一下（iOS 上更有感；系統若合併則等效單下）
  setTimeout(function(){
    if(t){try{t.click();}catch(e){}}
    try{if(navigator.vibrate)navigator.vibrate(10);}catch(e){}
  },40);
}

function bioSupported(){
  return !!(window.isSecureContext&&navigator.credentials&&navigator.credentials.create
    &&navigator.credentials.get&&window.PublicKeyCredential!==undefined);
}
function b64urlToBytes(s){
  const b=atob(s.replace(/-/g,'+').replace(/_/g,'/'));
  const u=new Uint8Array(b.length);
  for(let i=0;i<b.length;i++)u[i]=b.charCodeAt(i);
  return u;
}
function bytesToB64url(u){
  let s='';const chunk=0x8000;
  for(let i=0;i<u.length;i+=chunk)s+=String.fromCharCode.apply(null,u.subarray(i,i+chunk));
  return btoa(s).replace(/\+/g,'-').replace(/\//g,'_').replace(/=+$/,'');
}

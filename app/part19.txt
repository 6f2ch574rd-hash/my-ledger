async function importJSON(){
  let text;try{text=await readFile($('jsonFile'));}catch(e){toast('請先選擇 JSON 檔案');return;}
  let pack;try{pack=JSON.parse(text);}catch(e){toast('檔案格式不正確');return;}
  if(pack.app!=='my-ledger'||!pack.db){toast('不是本 App 的備份檔');return;}
  if(!confirm('還原會覆蓋這支手機上目前的所有記錄，確定繼續？建議先匯出目前備份。'))return;
  books=pack.books||books;DB=pack.db;activeId=pack.activeBook||books[0].id;
  if(!DB[activeId])DB[activeId]=freshBookData();
  if(pack.settings&&pack.settings.pass!==undefined)settings.pass=pack.settings.pass;
  saveBooks();saveDB();LS.set('ll_settings',settings);
  refreshAll();toast('✅ 備份已還原');
}

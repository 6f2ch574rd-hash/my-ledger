function renderDataSafety(){
  // 永久儲存狀態
  const ps=$('persistStatus');
  if(navigator.storage&&navigator.storage.persisted){
    navigator.storage.persisted().then(granted=>{
      ps.textContent=granted?'✅ 已取得永久儲存權限，系統不會在容量不足時自動清除資料':'⏳ 尚未取得永久儲存權限（用 Safari 開啟並加入主畫面可提升保存等級）';
    }).catch(()=>{ps.textContent='';});
  }else ps.textContent='';
  // 備份提醒狀態
  const last=LS.get('ll_lastBackup',0);
  const bs=$('backupStatus');
  if(!last){bs.innerHTML='📦 尚未備份過，建議現在匯出一次完整備份。';bs.style.color='#b07a00';}
  else{
    const days=Math.floor((nowTs()-last)/86400000);
    const d=new Date(last);
    const dstr=`${d.getFullYear()}-${String(d.getMonth()+1).padStart(2,'0')}-${String(d.getDate()).padStart(2,'0')}`;
    bs.innerHTML=`🕘 上次備份：${dstr}（${days} 天前）`;
    bs.style.color=days>30?'#c0392b':'#888';
  }
}

}

function initLock(){
  if(!settings.pass)return;

  let buf='';

  const keys=['1','2','3','4','5','6','7','8','9','C','0','⌫'];
  $('lockPad').innerHTML=keys.map(k=>`<button class="${k==='C'?'sub':''}" data-k="${k}">${k}</button>`).join('');
  $('lockPad').onclick=e=>{const k=e.target.dataset.k;if(!k)return;
    haptic();
    if(k==='C')buf='';else if(k==='⌫')buf=buf.slice(0,-1);else if(buf.length<4)buf+=k;
    draw();
    if(buf.length===4)setTimeout(()=>{
      if(buf===settings.pass){$('lockScreen').classList.remove('show');}
      else{buf='';draw();toast('密碼錯誤');}
    },120);
  };
  if(settings.bio&&settings.bioCred&&bioSupported()){
    setTimeout(()=>doBioUnlock(true),350);
  }
}

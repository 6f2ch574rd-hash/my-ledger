  if(!settings.pass)return;
  $('lockScreen').classList.add('show');
  $('faceBtn').style.display=(settings.bio&&settings.bioCred&&bioSupported())?'block':'none';
  let buf='';
  const draw=()=>[...$('lockDots').children].forEach((d,i)=>d.className=buf[i]?'fill':'');
  const keys=['1','2','3','4','5','6','7','8','9','C','0','⌫'];
  $('lockPad').innerHTML=keys.map(k=>`<button class="${k==='C'?'sub':''}" data-k="${k}">${k}</button>`).join('');
  $('lockPad').onclick=e=>{const k=e.target.dataset.k;if(!k)return;
    haptic();
    if(k==='C')buf='';else if(k==='⌫')buf=buf.slice(0,-1);else if(buf.length<4)buf+=k;
    draw();
    if(buf.length===4)setTimeout(()=>{
      if(buf===settings.pass){$('lockScreen').classList.remove('show');}
      else{buf='';draw();toast('密碼錯誤');}
    },120);
  };
  if(settings.bio&&settings.bioCred&&bioSupported()){
    setTimeout(()=>doBioUnlock(true),350);
  }
}

function download(name,content,mime,bom){
  const blob=new Blob([bom?'﻿':'',content],{type:mime});
  const a=document.createElement('a');a.href=URL.createObjectURL(blob);a.download=name;a.click();

}
function csvParse(text){
  if(text.charCodeAt(0)===0xFEFF)text=text.slice(1);
  const rows=[];let cur=[''],q=false;
  for(let i=0;i<text.length;i++){const c=text[i];
    if(q){if(c==='"'){if(text[i+1]==='"'){cur[cur.length-1]+='"';i++;}else q=false;}else cur[cur.length-1]+=c;}
    else if(c==='"')q=true;
    else if(c===',')cur.push('');
    else if(c==='\n'||c==='\r'){if(c==='\r'&&text[i+1]==='\n')i++;rows.push(cur);cur=[''];}
    else cur[cur.length-1]+=c;}
  if(cur.length>1||cur[0]!=='')rows.push(cur);
  return rows;
}

function addAccount(){
  haptic();
  const name=$('newAcctName').value.trim(),icon=$('newAcctIcon').value.trim()||'💳',open=Number($('newAcctOpen').value)||0;
  if(!name){toast('請輸入帳戶名稱');return;}
  D().accounts.push({id:uid(),name,icon,opening:open});saveDB();renderSettings();toast('已新增帳戶');
}
function delAccount(i){
  haptic();
  const a=D().accounts[i];
  if(D().tx.some(r=>r.account===a.id)){toast('這個帳戶已有記錄，建議保留');return;}
  if(!confirm('刪除帳戶「'+a.name+'」？'))return;
  D().accounts.splice(i,1);saveDB();renderSettings();
}
let catMgrType='expense';
function switchCatMgr(t,btn){
  haptic();
  catMgrType=t;
  btn.parentElement.children[0].className=t==='expense'?'on-exp':'';
  btn.parentElement.children[1].className=t==='income'?'on-inc':'';
  btn.parentElement.classList.toggle('inc',t==='income');
  renderCatMgrList();
}
function renderCatMgrList(){
  const list=catMgrType==='expense'?D().catsExp:D().catsInc;
  $('catMgrBox').innerHTML=list.map((c,i)=>
    `<span class="chip" style="padding-right:4px;">${esc(c)} <button class="del-x" style="font-size:13px;padding:0 4px;display:inline;" onclick="delCategory(${i})">×</button></span>`).join('');
}
function addCategory(){
  haptic();
  const name=$('newCatName').value.trim(),emoji=$('newCatEmoji').value.trim();
  if(!name){toast('請輸入分類名稱');return;}
  const full=(emoji?emoji+' ':'')+name;
  const arr=catMgrType==='expense'?D().catsExp:D().catsInc;
  if(arr.includes(full)){toast('分類已存在');return;}
  arr.push(full);saveDB();$('newCatName').value='';$('newCatEmoji').value='';renderCatMgrList();toast('已新增分類');
}
function delCategory(i){
  haptic();
  const arr=catMgrType==='expense'?D().catsExp:D().catsInc;
  if(arr.length<=1){toast('至少保留一個分類');return;}
  arr.splice(i,1);saveDB();renderCatMgrList();
}
